# AI Player MCP Bridge - Diplomacy Dump
# Exports diplomatic relations to game.log

ai_player_dump_diplomacy = {
	every_relation = {
		limit = {
			is_country_type = default
			NOT = { is_same_value = root }
		}

		root = {
			export_trigger_value_to_variable = {
				trigger = opinion
				parameters = {
					who = prev
				}
				variable = ai_dip_opinion
			}
		}

		# Triggers are not localization properties. Export their evaluated values
		# as 0/1, scoped to the relation country and specifically against Root.
		set_variable = { which = ai_dip_rival value = 0 }
		set_variable = { which = ai_dip_pact value = 0 }
		set_variable = { which = ai_dip_war value = 0 }
		if = {
			limit = { is_rival = root }
			set_variable = { which = ai_dip_rival value = 1 }
		}
		if = {
			limit = { has_defensive_pact = root }
			set_variable = { which = ai_dip_pact value = 1 }
		}
		if = {
			limit = { is_at_war_with = root }
			set_variable = { which = ai_dip_war value = 1 }
		}

		log = "AI_DIPLO|\\[This.GetName]|\\[Root.ai_dip_opinion]|\\[This.ai_dip_rival]|\\[This.ai_dip_pact]|\\[This.ai_dip_war]"
		clear_variable = ai_dip_rival
		clear_variable = ai_dip_pact
		clear_variable = ai_dip_war
	}
}
