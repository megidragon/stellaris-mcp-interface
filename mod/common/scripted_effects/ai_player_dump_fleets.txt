# AI Player MCP Bridge - Fleet Dump
# Exports owned fleet data to game.log

ai_player_dump_fleets = {
	every_owned_fleet = {
		limit = {
			is_mobile = yes
		}

		export_trigger_value_to_variable = {
			trigger = num_ships
			variable = ai_fl_ships
		}
		export_trigger_value_to_variable = {
			trigger = fleet_power
			variable = ai_fl_power
		}

		# No documented fleet MIA localization/trigger is available. Report it
		# explicitly as unknown rather than silently treating missing data as no.
		log = "AI_FLEET|\\[This.GetName]|\\[This.ai_fl_ships]|\\[This.ai_fl_power]|unknown"
	}
}
