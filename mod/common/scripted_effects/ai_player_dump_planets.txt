# AI Player MCP Bridge - Planet Dump
# Exports owned planet data to game.log

ai_player_dump_planets = {
	every_owned_planet = {
		export_trigger_value_to_variable = {
			trigger = count_owned_pop_amount
			variable = ai_pl_pops
		}
		export_trigger_value_to_variable = {
			trigger = planet_size
			variable = ai_pl_size
		}
		export_trigger_value_to_variable = {
			trigger = planet_stability
			variable = ai_pl_stability
		}
		export_trigger_value_to_variable = {
			trigger = free_amenities
			variable = ai_pl_amenities
		}
		export_trigger_value_to_variable = {
			trigger = free_housing
			variable = ai_pl_free_housing
		}
		export_trigger_value_to_variable = {
			trigger = free_jobs
			variable = ai_pl_free_jobs
		}
		export_trigger_value_to_variable = {
			trigger = planet_crime
			variable = ai_pl_crime
		}
		export_trigger_value_to_variable = {
			trigger = num_buildings
			parameters = {
				type = any
				value = 0
			}
			variable = ai_pl_buildings
		}
		export_trigger_value_to_variable = {
			trigger = num_districts
			parameters = {
				type = any
				value = 0
			}
			variable = ai_pl_districts
		}

		log = "AI_PLANET|\\[This.GetName]|\\[This.GetClassName]|\\[This.ai_pl_pops]|\\[This.ai_pl_size]|\\[This.ai_pl_stability]|\\[This.ai_pl_amenities]|\\[This.ai_pl_free_housing]|\\[This.ai_pl_free_jobs]|\\[This.ai_pl_crime]|\\[This.ai_pl_buildings]|\\[This.ai_pl_districts]"
	}
}
