# AI Player MCP Bridge - Resource Dump
# Exports resource stockpiles and net monthly flow to game.log.
# resource_income_compare is documented as profit minus loss, not revenue.
# The trigger export syntax is also used by vanilla cyber.5016 in
# events/machine_age_events_2.txt. Keep AI_RESOURCES fields unchanged.

ai_player_dump_resources = {
	# Energy
	export_resource_stockpile_to_variable = {
		resource = energy
		variable = ai_res_energy_s
	}
	export_trigger_value_to_variable = {
		trigger = resource_income_compare
		parameters = { resource = energy }
		variable = ai_res_energy_i
	}
	log = "AI_RESOURCES|\\[GetDate]|energy|\\[This.ai_res_energy_s]|\\[This.ai_res_energy_i]"
